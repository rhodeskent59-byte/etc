function asInput(input){
  return (input && typeof input==="object" && !Array.isArray(input)) ? input : {};

function findVal(input, base){
  if(input==null) return undefined;
  if(Object.prototype.hasOwnProperty.call(input, base)) return input[base];
  for(const k of Object.keys(input)){ if(k.startsWith(base+"@")) return input[k]; }
  return undefined;
}

  const inp=asInput(input); const out=[];
  if(Object.prototype.hasOwnProperty.call(inp, base)) out.push(inp[base]);
  for(const k of Object.keys(inp)){ if(k.startsWith(base+"@")) out.push(inp[k]); }
  return out;
}
function plausibleNums(input, base){
  const range=PLAUSIBLE_RANGES[base];
  return readingValues(input, base).filter(v=>typeof v==="number" && Number.isFinite(v) && (!range || (v>=range[0] && v<=range[1])));
}
// Aggregate a numeric datapoint across all its sensors/zones, using only plausible values.
// mode: "min" (coldest / lowest — the risk end for freeze, setpoint lag, battery), "max" (hottest /
// stalest / highest — the risk end for overload, CO2, offline), or "first" (legacy single-sensor view).
function aggregateReading(input, base, mode){
  const vals=plausibleNums(input, base);
  if(!vals.length) return undefined;
  if(mode==="min") return Math.min(...vals);
  if(mode==="max") return Math.max(...vals);
  return vals[0];
}
function anyEquals(input, base, wanted){
  return readingValues(input, base).some(v=>v===wanted);
}
function saneCount(v){ return (typeof v==="number" && Number.isFinite(v) && v>=0) ? v : undefined; }
function makeFault(def, input){
  const snapshotValues={};
  for(const s of def.snapshot){ snapshotValues[s]="see readings"; }
  // Fill concrete values where we can map — using the same aggregates the triggers use, so the snapshot
  // a technician receives shows the value that actually fired the fault, not a different sensor's value.
  const map={smoke_alarm_state:["smoke_alarm_state","first"],co_alarm_state:["co_alarm_state","first"],zone_temp:["zone_temp","min"],pipe_zone_temp:["pipe_zone_temp","min"],water_flow:["water_flow","first"],circuit_load_pct:["circuit_load_pct","max"],whole_home_power:["whole_home_power","first"],plug_power:["plug_power","max"],hvac_starts_per_hour:["hvac_starts_per_hour","first"],hvac_delta_t:["hvac_delta_t","first"],hvac_supply_temp:["hvac_supply_temp","first"],hvac_return_temp:["hvac_return_temp","first"],co2:["co2","max"],zone_humidity:["zone_humidity","first"],radon:["radon","max"],sensor_last_seen:["sensor_last_seen","max"],sensor_battery:["sensor_battery","min"]};
  for(const s of def.snapshot){
    for(const base in map){ if(s.includes(base)){ const v=(base==="smoke_alarm_state"||base==="co_alarm_state"||base==="water_flow") ? findVal(input, base) : aggregateReading(input, map[base][0], map[base][1]); if(v!==undefined) snapshotValues[s]=v; } }
    if(s.includes("leak_detected")){ if(anyEquals(input,"leak_detected","wet")) snapshotValues[s]="wet"; else { const v=findVal(input,"leak_detected"); if(v!==undefined) snapshotValues[s]=v; } }
    if(s.includes("nightly baseline")){ const b=asInput(input).whole_home_power_nightly_baseline_w; if(b && typeof b==="object") snapshotValues[s]=b; }
    if(s.includes("service dates")){ snapshotValues[s]={days_since_filter_change:asInput(input).days_since_filter_change, filter_interval_days:asInput(input).filter_interval_days}; }
    if(s.includes("last readings")){ snapshotValues[s]="last readings captured at scan time"; }
  }
  return {...def, snapshotValues, status:"active", firedAt:new Date().toISOString()};
}
function healthFromFaults(faults){
  if(!Array.isArray(faults) || faults.length===0) return "OK";
  if(faults.some(f=>f && f.severity==="ACT NOW")) return "ACT NOW";
  if(faults.some(f=>f && f.severity==="SCHEDULE")) return "SCHEDULE";
  if(faults.some(f=>f && f.severity==="WATCH")) return "WATCH";
  return "OK";
}
function evaluateFaults(rawInput){
  const input=asInput(rawInput);
  const faults=[]; const byCode={}; for(const d of FAULT_CODES) byCode[d.code]=d;
  const add=(code)=>{ if(!faults.some(f=>f.code===code)) faults.push(makeFault(byCode[code], input)); };
  const pipe=aggregateReading(input,"pipe_zone_temp","min");
  const circuit=aggregateReading(input,"circuit_load_pct","max");
  const co2=aggregateReading(input,"co2","max");
  const humidityDur=saneCount(input.humidity_duration_hr);
  // S — life safety: an alarm state at ANY detector fires the code.
  if(anyEquals(input,"smoke_alarm_state","alarm")) add("S-001");
  if(anyEquals(input,"co_alarm_state","alarm")) add("S-002");
  if(typeof pipe==="number" && pipe < 34) add("S-101");
  else if(typeof pipe==="number" && pipe < 38) add("S-102");
  // P — a wet report at ANY leak sensor fires P-001, even if another leak sensor reports dry.
  if(anyEquals(input,"leak_detected","wet")) add("P-001");
  const flow=findVal(input,"water_flow"); const flowDur=saneCount(input.flow_duration_min) ?? 0;
  if(flow==="flowing" && input.away_state==="on" && flowDur>=10) add("P-002");
  if(flow==="flowing" && input.away_state!=="on" && flowDur>=60) add("P-101");
  // Humidity is judged per zone entry: basement/mechanical entries feed P-201, other zones feed A-102.
  const humidityEntries=[];
  for(const k of Object.keys(input)){
    if(k==="zone_humidity" || k.startsWith("zone_humidity@")){
      const v=input[k]; if(typeof v!=="number" || !Number.isFinite(v) || v<0 || v>100) continue;
      const zone=k.includes("@") ? k.slice(k.indexOf("@")+1) : (typeof input.humidity_zone==="string" ? input.humidity_zone : "");
      humidityEntries.push({value:v, basement:/basement|mechanical/i.test(zone)});
    }
  }
  if(humidityEntries.some(e=>e.basement && e.value>60) && (humidityDur===undefined || humidityDur>=24)) add("P-201");
  if(humidityEntries.some(e=>!e.basement && (e.value<30 || e.value>60)) && typeof humidityDur==="number" && humidityDur>=24) add("A-102");
  // E
  if(typeof circuit==="number" && circuit>100) add("E-001");
  if(typeof circuit==="number" && circuit>80 && circuit<=100 && (saneCount(input.load_duration_min) ?? 0)>=30) add("E-101");
  // Plug load: power is only compared with the limit belonging to the SAME plug (same @suffix).
  {
    const suffixes=new Set();
    for(const k of Object.keys(input)){
      if(k==="plug_power" || k.startsWith("plug_power@")) suffixes.add(k.includes("@") ? k.slice(k.indexOf("@")+1) : "");
      if(k==="plug_limit_w" || k.startsWith("plug_limit_w@") || k==="plug_limit" || k.startsWith("plug_limit@")) suffixes.add(k.includes("@") ? k.slice(k.indexOf("@")+1) : "");
    }
    for(const suf of suffixes){
      const pKey=suf ? "plug_power@"+suf : "plug_power";
      const lKey=(suf ? "plug_limit_w@"+suf : "plug_limit_w") in input ? (suf ? "plug_limit_w@"+suf : "plug_limit_w") : (suf ? "plug_limit@"+suf : "plug_limit");
      const p=input[pKey], l=input[lKey];
      if(typeof p==="number" && Number.isFinite(p) && p>=0 && p<=20000 && typeof l==="number" && Number.isFinite(l) && l>=0 && l<=20000 && p>l){ add("E-102"); break; }
    }
  }
  const base=input.whole_home_power_nightly_baseline_w;
  if(base && typeof base==="object" && Number.isFinite(base.nights_1_7_avg) && Number.isFinite(base.nights_8_14_avg) && base.nights_1_7_avg>=0 && base.nights_8_14_avg>=0 && base.nights_8_14_avg > base.nights_1_7_avg*1.2 && (base.nights_8_14_avg-base.nights_1_7_avg)>50) add("E-201");
  // H
  const starts=aggregateReading(input,"hvac_starts_per_hour","first");
  if(typeof starts==="number" && starts>6) add("H-001");
  const delta=aggregateReading(input,"hvac_delta_t","first");
  if(typeof delta==="number" && delta<25 && (input.burner_state==="running" || input.burner_state==="on")) add("H-002");
  const zoneTemp=aggregateReading(input,"zone_temp","min");
  const setpoint=aggregateReading(input,"setpoint","first");
  if(typeof zoneTemp==="number" && typeof setpoint==="number" && input.heating_call==="active" && (setpoint-zoneTemp)>=5 && ((saneCount(input.duration_min) ?? 0)>=60)) add("H-101");
  const daysFilter=saneCount(input.days_since_filter_change); const filterInterval=saneCount(input.filter_interval_days);
  if(typeof daysFilter==="number" && typeof filterInterval==="number" && filterInterval>0 && daysFilter>filterInterval) add("H-201");
  // A
  const radon=aggregateReading(input,"radon","max");
  if(typeof radon==="number" && radon>4.0) add("A-001");
  if(typeof co2==="number" && co2>1000 && (input.co2_duration_min===undefined || (saneCount(input.co2_duration_min) ?? 0)>=30)) add("A-101");
  // N — the stalest sensor and the lowest battery are the risk ends.
  const lastSeen=aggregateReading(input,"sensor_last_seen","max");
  if(typeof lastSeen==="number" && lastSeen>2) add("N-001");
  const batt=aggregateReading(input,"sensor_battery","min");
  if(typeof batt==="number" && batt<20) add("N-002");
  const hb=input.gateway_heartbeat;
  if(hb==="absent beyond design-default window" || hb==="absent" || hb===false || hb==="offline") add("N-003");
  return faults;
}
function runScan(input, notMonitored){
  const faults=evaluateFaults(input);
  return {faults, faultCount:faults.length, codes:faults.map(f=>f.code), healthState:healthFromFaults(faults), notMonitored: Array.isArray(notMonitored) ? notMonitored : NOT_MONITORED_DEFAULT, scannedAt:new Date().toISOString()};
}

// onboarding state, and the SQLite-shaped export/import for the standalone app.

//  - A report is only ever built from a scan OF THE SAME BUILDING (latestScanForBuilding never

//  - A building with no sensors is NOT MONITORED (scanHealthDisplay), never silently "OK".

//    the caller's store is left untouched.

// Normalize any migratable store (v0.1.x / v0.2.x legacy shapes or v0.3.x) into a valid v0.3 store.
// Throws on structurally invalid records; pushes human-readable warnings for coerced records.
function normalizeStore(raw, warnings){
  const warn=(m)=>{ if(Array.isArray(warnings)) warnings.push(m); };
  if(!isObj(raw)) throw new Error("That backup does not contain a Building OBD store — refusing to import it. Nothing was changed.");
  for(const k of ["faults","scans","workOrders","reports"]){
    if(raw[k]!==undefined && !Array.isArray(raw[k])) throw new Error(`That backup's ${k} record is the wrong type — refusing to import it. Nothing was changed.`);
  }
  if(raw.buildings!==undefined && !Array.isArray(raw.buildings)) throw new Error("That backup's buildings record is the wrong type — refusing to import it. Nothing was changed.");
  let buildings;
  if(Array.isArray(raw.buildings) && raw.buildings.length){
    buildings=raw.buildings.map((b,i)=>sanitizeBuilding(b,i));
  } else {
    // v0.1.x legacy shape: one implicit building (the Lowell test building) with top-level profileDates.
    const b=defaultBuilding();
    if(isObj(raw.profileDates)){
      b.profileDates={ filterChanged: saneDate(raw.profileDates.filterChanged), filterIntervalDays: saneInterval(raw.profileDates.filterIntervalDays), detectorTested: saneDate(raw.profileDates.detectorTested), hvacServiced: saneDate(raw.profileDates.hvacServiced) };
    }
    buildings=[b];
  }
  const ids=new Set();
  for(const b of buildings){ if(ids.has(b.id)) b.id=b.id+"-"+ids.size; ids.add(b.id); }
  const fallbackId=buildings.some(b=>b.id===DEFAULT_BUILDING_ID) ? DEFAULT_BUILDING_ID : buildings[0].id;
  const tagRecords=(list)=>{
    if(!Array.isArray(list)) return [];
    return list.filter(isObj).map(r=>{
      const out={...r};
      if(typeof out.buildingId!=="string" || !ids.has(out.buildingId)) out.buildingId=fallbackId;
      if((out.code==="S-001" || out.code==="S-002") && out.status==="resolved"){
        out.status="acknowledged";
        out.resolutionNote="Imported record claimed this life-safety code was resolved from a screen. Life-safety codes resolve only at the device itself, so the record was reset to acknowledged on import.";
        warn(`${out.code} in an imported backup was marked resolved from a screen — reset to acknowledged. Life-safety codes resolve only at the device.`);
      }
      return out;
    });
  };
  const active=(typeof raw.activeBuildingId==="string" && ids.has(raw.activeBuildingId)) ? raw.activeBuildingId : fallbackId;
  return {
    version: APP_VERSION,
    activeBuildingId: active,
    buildings,
    faults: tagRecords(raw.faults),
    scans: tagRecords(raw.scans),
    workOrders: tagRecords(raw.workOrders),
    reports: tagRecords(raw.reports),
    settings: { units: (isObj(raw.settings) && raw.settings.units==="C") ? "C" : "F" },
    simulator: raw.simulator!==false
  };
}
// Migrate a v0.1.x / v0.2.x store object to v0.3. A v0.3.x store is simply normalized.
function migrateLegacyStore(raw){ return normalizeStore(raw); }
// Validate a backup (JSON text or already-parsed value). A backup may be either the browser-store
// JSON or the SQLite-shaped JSON export (format: "building-obd-sqlite-json") — both are accepted.
// Returns {ok:true, store, warnings} or {ok:false, error}. On any failure the caller's existing
// store must be left untouched.
function validateBackup(data){
  let obj=data;
  if(typeof data==="string"){
    try{ obj=JSON.parse(data); }catch(e){ return { ok:false, error:"That file is not valid JSON — it is not a readable Building OBD backup. Nothing was changed." }; }
  }
  if(!isObj(obj)) return { ok:false, error:"That file does not contain a Building OBD store object. Nothing was changed." };
  if(obj.format==="building-obd-sqlite-json") return importSqliteShape(obj);
  if(typeof obj.version!=="string" || !/^\d+\.\d+\.\d+$/.test(obj.version)) return { ok:false, error:"That backup has no recognisable version — refusing to import it. Nothing was changed." };
  const parts=obj.version.split(".").map(Number);
  if(parts[0]>0 || parts[1]>3) return { ok:false, error:`That backup is from a newer app version (${obj.version}) than this build (${APP_VERSION}) — refusing to import it. Nothing was changed.` };
  const warnings=[];
  try{
    const store=normalizeStore(obj, warnings);
    return { ok:true, store, warnings };
  }catch(e){
    return { ok:false, error:e.message };
  }
}
function addBuildingToStore(store, info){
  const name=isObj(info) && typeof info.name==="string" ? info.name.trim() : "";
  if(!name) return { ok:false, error:"A new building needs at least a name — the button did act, it just needs a name." };
  const address=isObj(info) && typeof info.address==="string" ? info.address.trim() : "";
  let id="b-"+(name.toLowerCase().replace(/[^a-z0-9]+/g,"-").replace(/^-+|-+$/g,"") || "building");
  if(store.buildings.some(b=>b.id===id)){ let n=2; while(store.buildings.some(b=>b.id===id+"-"+n)) n++; id=id+"-"+n; }
  const building={
    id, name, address, facts: null, simulator: false, sensors: [],
    notMonitored: [...NOT_MONITORED_DEFAULT, "All building systems — no sensors registered for this building yet"],
    profileDates: emptyProfileDates(),
    membership: { tier:"none", source:"manual", note:"", setAt: null },
    onboarding: { kitId: null, survey: {} },
    techChecks: {}
  };
  store.buildings.push(building);
  return { ok:true, building };
}

// Records belonging to exactly one building — faults, scans, work orders and reports never leak across buildings.
function recordsForBuilding(store, buildingId){
  const pick=(list)=>list.filter(r=>recordBuildingId(store,r)===buildingId);
  return { faults: pick(store.faults), scans: pick(store.scans), workOrders: pick(store.workOrders), reports: pick(Array.isArray(store.reports)?store.reports:[]) };
}

  const scans=recordsForBuilding(store, buildingId).scans;
  return scans.length ? scans[scans.length-1] : null;

// Merge newly scanned faults into the store, deduplicated per building: the same code active in
// another building never blocks (or is blocked by) this building's record.
function mergeFaultsForBuilding(store, buildingId, newFaults){
  let added=0;
  for(const nf of (Array.isArray(newFaults) ? newFaults : [])){
    const dup=store.faults.some(f=>recordBuildingId(store,f)===buildingId && f.code===nf.code && (f.status==="active" || f.status==="acknowledged"));
    if(!dup){ store.faults.push({...nf, buildingId, simulator: nf.simulator!==undefined ? nf.simulator : store.simulator}); added++; }
  }
  return added;
}

      };

// Building OBD - Gateway Layer (single source of truth) — v0.3.0
// Adapter interface between the app and whatever brings sensor readings in.
// index.html inlines this file (import/export stripped) exactly as it does the fault engine
// and the store; tests import it directly. Pure logic only — no network calls in here.
//
// Two adapters ship in this build:
//  (a) SIMULATOR adapter — feeds the Lowell test building from the in-app simulator registry.
//      It is badged SIMULATOR everywhere and never pretends to be a real hub.
//  (b) Real-transport adapter SKELETON for HTTP / MQTT polling — ships DISABLED. Until it is
//      enabled, configured, AND a transport actually answers with readings, the honest state is:
//        - disabled or unconfigured  -> "NOT CONNECTED"
//        - configured, enabled, but silent / erroring -> "OFFLINE" (never fake-live)
//      Readings are only ever returned when the injected responder actually returned them.
//      Nothing in this module invents, estimates, or replays a reading as if it were live.
const GATEWAY_NOT_CONNECTED = "NOT CONNECTED";
const GATEWAY_OFFLINE = "OFFLINE";
const GATEWAY_ONLINE = "ONLINE";
function validateGatewayConfig(config){
  const problems=[];
  const c=(config && typeof config==="object") ? config : {};
  if(c.transport!=="http" && c.transport!=="mqtt") problems.push("transport must be 'http' or 'mqtt'");
  if(c.transport==="http"){
    if(typeof c.url!=="string" || !/^https?:\/\/.+/.test(c.url)) problems.push("an HTTP gateway needs a poll URL (http:// or https://...)");
  }
  if(c.transport==="mqtt"){
    if(typeof c.broker!=="string" || !/^(mqtts?|wss?):\/\/.+/.test(c.broker)) problems.push("an MQTT gateway needs a broker URL (mqtt://, mqtts://, ws:// or wss://...)");
    if(typeof c.topic!=="string" || !c.topic.trim()) problems.push("an MQTT gateway needs at least one topic to subscribe to");
  }
  if(c.timeoutMs!==undefined && !(typeof c.timeoutMs==="number" && Number.isFinite(c.timeoutMs) && c.timeoutMs>0)) problems.push("timeoutMs must be a positive number when given");
  return problems;
}
function createSimulatorAdapter(getReadings){
  let lastPollAt=null; let lastReadings=null;
  return {
    kind: "simulator",
    badge: "SIMULATOR",
    isSimulator: true,
    enabled: true,
    configured: true,
    state(){ return GATEWAY_ONLINE; },
    poll(nowMs){
      const at=(typeof nowMs==="number") ? nowMs : Date.now();
      let readings=null;
      try{ readings=(typeof getReadings==="function") ? getReadings() : null; }catch(e){ readings=null; }
      lastPollAt=at; lastReadings=readings;
      return { state: GATEWAY_ONLINE, readings, at, simulator: true };
    },
    describe(){ return "Simulator gateway: ONLINE (SIMULATOR) — readings come from the in-app simulator registry, badged SIMULATOR, never presented as a real hub."; }
  };
}
function createRealAdapter(config, responder){
  const cfg=(config && typeof config==="object") ? config : {};
  const timeoutMs=(typeof cfg.timeoutMs==="number" && Number.isFinite(cfg.timeoutMs) && cfg.timeoutMs>0) ? cfg.timeoutMs : 10000;
  let lastSuccessAt=null; let lastAttemptAt=null; let lastError=null;
  const adapter={
    kind: (cfg.transport==="mqtt") ? "mqtt" : "http",
    badge: "REAL",
    isSimulator: false,
    enabled: cfg.enabled===true,
    configured: validateGatewayConfig(cfg).length===0,
    configProblems(){ return validateGatewayConfig(cfg); },
    state(nowMs){
      if(!adapter.enabled || !adapter.configured) return GATEWAY_NOT_CONNECTED;
      if(lastSuccessAt===null) return GATEWAY_OFFLINE;
      const now=(typeof nowMs==="number") ? nowMs : Date.now();
      return (now-lastSuccessAt<=timeoutMs) ? GATEWAY_ONLINE : GATEWAY_OFFLINE;
    },
    poll(nowMs){
      const now=(typeof nowMs==="number") ? nowMs : Date.now();
      if(!adapter.enabled) return { state: GATEWAY_NOT_CONNECTED, readings: null, at: now, reason: "adapter disabled — the real-transport adapter ships disabled in this build" };
      if(!adapter.configured) return { state: GATEWAY_NOT_CONNECTED, readings: null, at: now, reason: "adapter not configured: "+validateGatewayConfig(cfg).join("; ") };
      lastAttemptAt=now;
      let readings=null;
      try{ readings=(typeof responder==="function") ? responder(cfg) : null; }catch(e){ lastError=(e && e.message) || "transport error"; readings=null; }
      if(readings && typeof readings==="object" && !Array.isArray(readings) && Object.keys(readings).length){
        lastSuccessAt=now; lastError=null;
        return { state: GATEWAY_ONLINE, readings, at: now };
      }
      // Silent or erroring transport: OFFLINE, and readings stay null — never fake-live.
      return { state: GATEWAY_OFFLINE, readings: null, at: now, reason: lastError ? ("transport error: "+lastError) : "gateway configured but silent — no readings received" };
    },
    describe(){
      if(!adapter.enabled) return "Real gateway: NOT CONNECTED — the real-transport (HTTP/MQTT) adapter ships DISABLED in this build. No real hub is paired.";
      if(!adapter.configured) return "Real gateway: NOT CONNECTED — adapter enabled but not configured ("+validateGatewayConfig(cfg).join("; ")+").";
      return "Real gateway: "+adapter.state()+" ("+adapter.kind.toUpperCase()+" polling adapter).";
    }
  };
  return adapter;
}

function tick(){
  for(const s of SENSORS){ if(!s.online) continue; if(typeof s.base==="number"){ const cur=typeof readings[s.dp]==="number"?readings[s.dp]:s.base; let v=cur+(Math.random()*4-2); v=Math.max(s.min-5,Math.min(s.max+5,v)); readings[s.dp]=Math.round(v*10)/10; } else readings[s.dp]=s.base; }
  readings.hvac_delta_t=Math.round(((readings.hvac_supply_temp||118)-(readings.hvac_return_temp||71))*10)/10;
  readings.hvac_starts_per_hour=3; readings.water_flow="not flowing"; readings.whole_home_power=1840;
  // v0.2 fix: the scan-facing sensor-health values are computed from the registry, not hard-coded.
  // Battery = the lowest battery in the registry; last-seen jumps beyond the offline window when ANY
  // sensor is offline, so a normal scan honestly reports N-001 for a sensor the owner took offline.
  readings.sensor_battery=Math.min(...SENSORS.filter(s=>s.battery!==null).map(s=>s.battery));
  readings.sensor_last_seen=SENSORS.some(s=>!s.online)?3.5:0.4;
  const poll=simGateway.poll(); lastReadingAt=new Date(poll.at); renderStrip(); renderGauges();
}

function findVal(input, base){

// Every value reported for a datapoint: the base key plus every base@sensor / base@zone key.
export function readingValues(input, base){
  const inp=asInput(input); const out=[];
  if(Object.prototype.hasOwnProperty.call(inp, base)) out.push(inp[base]);

function plausibleNums(input, base){
  const range=PLAUSIBLE_RANGES[base];
  return readingValues(input, base).filter(v=>typeof v==="number" && Number.isFinite(v) && (!range || (v>=range[0] && v<=range[1])));
}
// Aggregate a numeric datapoint across all its sensors/zones, using only plausible values.
// mode: "min" (coldest / lowest — the risk end for freeze, setpoint lag, battery), "max" (hottest /
// stalest / highest — the risk end for overload, CO2, offline), or "first" (legacy single-sensor view).
export function aggregateReading(input, base, mode){
  const vals=plausibleNums(input, base);
  if(!vals.length) return undefined;
  if(mode==="min") return Math.min(...vals);
  if(mode==="max") return Math.max(...vals);
  return vals[0];
}

  if(anyEquals(input,"co_alarm_state","alarm")) add("S-002");

  else if(typeof pipe==="number" && pipe < 38) add("S-102");

  if(anyEquals(input,"leak_detected","wet")) add("P-001");

  if(flow==="flowing" && input.away_state==="on" && flowDur>=10) add("P-002");
  if(flow==="flowing" && input.away_state!=="on" && flowDur>=60) add("P-101");

  const humidityEntries=[];
  for(const k of Object.keys(input)){
    if(k==="zone_humidity" || k.startsWith("zone_humidity@")){

    }
  }
  if(humidityEntries.some(e=>e.basement && e.value>60) && (humidityDur===undefined || humidityDur>=24)) add("P-201");
  if(humidityEntries.some(e=>!e.basement && (e.value<30 || e.value>60)) && typeof humidityDur==="number" && humidityDur>=24) add("A-102");

// Building OBD - Gateway Layer (single source of truth) — v0.3.0
// Adapter interface between the app and whatever brings sensor readings in.
// index.html inlines this file (import/export stripped) exactly as it does the fault engine
// and the store; tests import it directly. Pure logic only — no network calls in here.
//
// Two adapters ship in this build:
//  (a) SIMULATOR adapter — feeds the Lowell test building from the in-app simulator registry.
//      It is badged SIMULATOR everywhere and never pretends to be a real hub.
//  (b) Real-transport adapter SKELETON for HTTP / MQTT polling — ships DISABLED. Until it is
//      enabled, configured, AND a transport actually answers with readings, the honest state is:
//        - disabled or unconfigured  -> "NOT CONNECTED"
//        - configured, enabled, but silent / erroring -> "OFFLINE" (never fake-live)
//      Readings are only ever returned when the injected responder actually returned them.
//      Nothing in this module invents, estimates, or replays a reading as if it were live.
export const GATEWAY_NOT_CONNECTED = "NOT CONNECTED";
export const GATEWAY_OFFLINE = "OFFLINE";
export const GATEWAY_ONLINE = "ONLINE";
export function validateGatewayConfig(config){
  const problems=[];
  const c=(config && typeof config==="object") ? config : {};
  if(c.transport!=="http" && c.transport!=="mqtt") problems.push("transport must be 'http' or 'mqtt'");
  if(c.transport==="http"){
    if(typeof c.url!=="string" || !/^https?:\/\/.+/.test(c.url)) problems.push("an HTTP gateway needs a poll URL (http:// or https://...)");
  }
  if(c.transport==="mqtt"){
    if(typeof c.broker!=="string" || !/^(mqtts?|wss?):\/\/.+/.test(c.broker)) problems.push("an MQTT gateway needs a broker URL (mqtt://, mqtts://, ws:// or wss://...)");
    if(typeof c.topic!=="string" || !c.topic.trim()) problems.push("an MQTT gateway needs at least one topic to subscribe to");
  }
  if(c.timeoutMs!==undefined && !(typeof c.timeoutMs==="number" && Number.isFinite(c.timeoutMs) && c.timeoutMs>0)) problems.push("timeoutMs must be a positive number when given");
  return problems;
}
export function createSimulatorAdapter(getReadings){
  let lastPollAt=null; let lastReadings=null;
  return {
    kind: "simulator",
    badge: "SIMULATOR",
    isSimulator: true,
    enabled: true,
    configured: true,
    state(){ return GATEWAY_ONLINE; },
    poll(nowMs){
      const at=(typeof nowMs==="number") ? nowMs : Date.now();
      let readings=null;
      try{ readings=(typeof getReadings==="function") ? getReadings() : null; }catch(e){ readings=null; }
      lastPollAt=at; lastReadings=readings;
      return { state: GATEWAY_ONLINE, readings, at, simulator: true };
    },
    describe(){ return "Simulator gateway: ONLINE (SIMULATOR) — readings come from the in-app simulator registry, badged SIMULATOR, never presented as a real hub."; }
  };
}
export function createRealAdapter(config, responder){
  const cfg=(config && typeof config==="object") ? config : {};
  const timeoutMs=(typeof cfg.timeoutMs==="number" && Number.isFinite(cfg.timeoutMs) && cfg.timeoutMs>0) ? cfg.timeoutMs : 10000;
  let lastSuccessAt=null; let lastAttemptAt=null; let lastError=null;
  const adapter={
    kind: (cfg.transport==="mqtt") ? "mqtt" : "http",
    badge: "REAL",
    isSimulator: false,
    enabled: cfg.enabled===true,
    configured: validateGatewayConfig(cfg).length===0,
    configProblems(){ return validateGatewayConfig(cfg); },
    state(nowMs){
      if(!adapter.enabled || !adapter.configured) return GATEWAY_NOT_CONNECTED;
      if(lastSuccessAt===null) return GATEWAY_OFFLINE;
      const now=(typeof nowMs==="number") ? nowMs : Date.now();
      return (now-lastSuccessAt<=timeoutMs) ? GATEWAY_ONLINE : GATEWAY_OFFLINE;
    },
    poll(nowMs){
      const now=(typeof nowMs==="number") ? nowMs : Date.now();
      if(!adapter.enabled) return { state: GATEWAY_NOT_CONNECTED, readings: null, at: now, reason: "adapter disabled — the real-transport adapter ships disabled in this build" };
      if(!adapter.configured) return { state: GATEWAY_NOT_CONNECTED, readings: null, at: now, reason: "adapter not configured: "+validateGatewayConfig(cfg).join("; ") };
      lastAttemptAt=now;
      let readings=null;
      try{ readings=(typeof responder==="function") ? responder(cfg) : null; }catch(e){ lastError=(e && e.message) || "transport error"; readings=null; }
      if(readings && typeof readings==="object" && !Array.isArray(readings) && Object.keys(readings).length){
        lastSuccessAt=now; lastError=null;
        return { state: GATEWAY_ONLINE, readings, at: now };
      }
      // Silent or erroring transport: OFFLINE, and readings stay null — never fake-live.
      return { state: GATEWAY_OFFLINE, readings: null, at: now, reason: lastError ? ("transport error: "+lastError) : "gateway configured but silent — no readings received" };
    },
    describe(){
      if(!adapter.enabled) return "Real gateway: NOT CONNECTED — the real-transport (HTTP/MQTT) adapter ships DISABLED in this build. No real hub is paired.";
      if(!adapter.configured) return "Real gateway: NOT CONNECTED — adapter enabled but not configured ("+validateGatewayConfig(cfg).join("; ")+").";
      return "Real gateway: "+adapter.state()+" ("+adapter.kind.toUpperCase()+" polling adapter).";
    }
  };
  return adapter;
}
export default { GATEWAY_NOT_CONNECTED, GATEWAY_OFFLINE, GATEWAY_ONLINE, validateGatewayConfig, createSimulatorAdapter, createRealAdapter };

// Normalize any migratable store (v0.1.x / v0.2.x legacy shapes or v0.3.x) into a valid v0.3 store.
// Throws on structurally invalid records; pushes human-readable warnings for coerced records.
export function normalizeStore(raw, warnings){
  const warn=(m)=>{ if(Array.isArray(warnings)) warnings.push(m); };
  if(!isObj(raw)) throw new Error("That backup does not contain a Building OBD store — refusing to import it. Nothing was changed.");
  for(const k of ["faults","scans","workOrders","reports"]){
    if(raw[k]!==undefined && !Array.isArray(raw[k])) throw new Error(`That backup's ${k} record is the wrong type — refusing to import it. Nothing was changed.`);
  }
  if(raw.buildings!==undefined && !Array.isArray(raw.buildings)) throw new Error("That backup's buildings record is the wrong type — refusing to import it. Nothing was changed.");
  let buildings;
  if(Array.isArray(raw.buildings) && raw.buildings.length){
    buildings=raw.buildings.map((b,i)=>sanitizeBuilding(b,i));
  } else {
    // v0.1.x legacy shape: one implicit building (the Lowell test building) with top-level profileDates.
    const b=defaultBuilding();
    if(isObj(raw.profileDates)){
      b.profileDates={ filterChanged: saneDate(raw.profileDates.filterChanged), filterIntervalDays: saneInterval(raw.profileDates.filterIntervalDays), detectorTested: saneDate(raw.profileDates.detectorTested), hvacServiced: saneDate(raw.profileDates.hvacServiced) };
    }
    buildings=[b];
  }
  const ids=new Set();
  for(const b of buildings){ if(ids.has(b.id)) b.id=b.id+"-"+ids.size; ids.add(b.id); }
  const fallbackId=buildings.some(b=>b.id===DEFAULT_BUILDING_ID) ? DEFAULT_BUILDING_ID : buildings[0].id;
  const tagRecords=(list)=>{
    if(!Array.isArray(list)) return [];
    return list.filter(isObj).map(r=>{
      const out={...r};
      if(typeof out.buildingId!=="string" || !ids.has(out.buildingId)) out.buildingId=fallbackId;
      if((out.code==="S-001" || out.code==="S-002") && out.status==="resolved"){
        out.status="acknowledged";
        out.resolutionNote="Imported record claimed this life-safety code was resolved from a screen. Life-safety codes resolve only at the device itself, so the record was reset to acknowledged on import.";
        warn(`${out.code} in an imported backup was marked resolved from a screen — reset to acknowledged. Life-safety codes resolve only at the device.`);
      }
      return out;
    });
  };
  const active=(typeof raw.activeBuildingId==="string" && ids.has(raw.activeBuildingId)) ? raw.activeBuildingId : fallbackId;
  return {
    version: APP_VERSION,
    activeBuildingId: active,
    buildings,
    faults: tagRecords(raw.faults),
    scans: tagRecords(raw.scans),
    workOrders: tagRecords(raw.workOrders),
    reports: tagRecords(raw.reports),
    settings: { units: (isObj(raw.settings) && raw.settings.units==="C") ? "C" : "F" },
    simulator: raw.simulator!==false
  };
}

// Validate a backup (JSON text or already-parsed value). A backup may be either the browser-store
// JSON or the SQLite-shaped JSON export (format: "building-obd-sqlite-json") — both are accepted.
// Returns {ok:true, store, warnings} or {ok:false, error}. On any failure the caller's existing
// store must be left untouched.
export function validateBackup(data){
  let obj=data;
  if(typeof data==="string"){
    try{ obj=JSON.parse(data); }catch(e){ return { ok:false, error:"That file is not valid JSON — it is not a readable Building OBD backup. Nothing was changed." }; }
  }
  if(!isObj(obj)) return { ok:false, error:"That file does not contain a Building OBD store object. Nothing was changed." };
  if(obj.format==="building-obd-sqlite-json") return importSqliteShape(obj);
  if(typeof obj.version!=="string" || !/^\d+\.\d+\.\d+$/.test(obj.version)) return { ok:false, error:"That backup has no recognisable version — refusing to import it. Nothing was changed." };
  const parts=obj.version.split(".").map(Number);
  if(parts[0]>0 || parts[1]>3) return { ok:false, error:`That backup is from a newer app version (${obj.version}) than this build (${APP_VERSION}) — refusing to import it. Nothing was changed.` };
  const warnings=[];
  try{
    const store=normalizeStore(obj, warnings);
    return { ok:true, store, warnings };
  }catch(e){
    return { ok:false, error:e.message };
  }
}

// Merge newly scanned faults into the store, deduplicated per building: the same code active in
// another building never blocks (or is blocked by) this building's record.
export function mergeFaultsForBuilding(store, buildingId, newFaults){
  let added=0;
  for(const nf of (Array.isArray(newFaults) ? newFaults : [])){
    const dup=store.faults.some(f=>recordBuildingId(store,f)===buildingId && f.code===nf.code && (f.status==="active" || f.status==="acknowledged"));
    if(!dup){ store.faults.push({...nf, buildingId, simulator: nf.simulator!==undefined ? nf.simulator : store.simulator}); added++; }
  }
  return added;
}

-- Building OBD — SQLite schema v1 (app v0.3.0, full build / Gate B candidate)
-- This is the real persistence schema behind the store interface.
-- The standalone HTML build keeps working offline on its browser store; this schema is the
-- shape that store takes when it leaves the browser:
--   exportSqliteShape(store)  ->  { format: "building-obd-sqlite-json", tables: { ... } }
--   importSqliteShape(data)   ->  validated store (refuses anything malformed)
-- Table and column names in the JSON export match this file exactly (snake_case columns).
-- Composite values (lists, snapshot maps) are stored as JSON text, named in the column comments.
-- Honesty constraint baked into the schema: memberships.source can only be 'manual' or 'comp'.
-- There is no 'paid' state in this build — the payment door is undecided, so no membership can
-- ever be recorded as paid. Life-safety codes (S-001/S-002) carry no special column: their
-- never-screen-clearable rule is enforced by the app and re-checked on every import.
--
-- Apply with:  sqlite3 building-obd.db < schema.sql        (or node:sqlite in the test suite)
-- Recommended: PRAGMA foreign_keys = ON;

CREATE TABLE IF NOT EXISTS buildings (
  id                  TEXT PRIMARY KEY,
  name                TEXT NOT NULL,
  address             TEXT,
  year_built          INTEGER,
  sqft                INTEGER,
  beds                INTEGER,
  baths               INTEGER,
  garage_sqft         INTEGER,
  lot                 TEXT,
  simulator           INTEGER NOT NULL DEFAULT 0,   -- 1 = simulator/test building (e.g. Lowell)
  not_monitored       TEXT NOT NULL DEFAULT '[]',   -- JSON array of system names, honest NOT MONITORED list
  filter_changed      TEXT,                          -- ISO date or NULL ("No record yet")
  filter_interval_days INTEGER,
  detector_tested     TEXT,
  hvac_serviced       TEXT,
  membership_tier     TEXT NOT NULL DEFAULT 'none', -- none | watch | health | full (denormalized copy of memberships)
  membership_source   TEXT NOT NULL DEFAULT 'manual' CHECK (membership_source IN ('manual','comp'))
);

CREATE TABLE IF NOT EXISTS sensors (
  building_id TEXT NOT NULL REFERENCES buildings(id),
  id          TEXT NOT NULL,
  name        TEXT,
  zone        TEXT,
  status      TEXT,                                 -- e.g. 'NOT CONNECTED', 'ONLINE (SIMULATOR)'
  simulator   INTEGER NOT NULL DEFAULT 0,
  PRIMARY KEY (building_id, id)
);

CREATE TABLE IF NOT EXISTS scans (
  id            TEXT PRIMARY KEY,                   -- export id: scan-<n> within one export
  building_id   TEXT NOT NULL REFERENCES buildings(id),
  scanned_at    TEXT,                               -- ISO timestamp from the scan record
  label         TEXT,
  health_state  TEXT,                               -- OK | WATCH | SCHEDULE | ACT NOW (display may still be NOT MONITORED)
  fault_count   INTEGER,
  codes         TEXT,                               -- JSON array of fault codes found by this scan
  simulator     INTEGER NOT NULL DEFAULT 0,
  not_monitored TEXT                                -- JSON array
);

CREATE TABLE IF NOT EXISTS readings (
  id          TEXT PRIMARY KEY,                     -- export id derived from the scan/fault/snapshot key
  scan_id     TEXT NOT NULL REFERENCES scans(id),
  building_id TEXT NOT NULL REFERENCES buildings(id),
  datapoint   TEXT,                                 -- snapshot key, e.g. leak_detected (which sensor)
  value       TEXT                                  -- snapshot value as text (JSON text when the value is composite)
);

CREATE TABLE IF NOT EXISTS faults (
  id            TEXT PRIMARY KEY,                   -- export id: fault-<n> within one export
  building_id   TEXT NOT NULL REFERENCES buildings(id),
  code          TEXT,                               -- library code, e.g. P-001
  name          TEXT,
  severity      TEXT,                               -- ACT NOW | SCHEDULE | WATCH
  status        TEXT,                               -- active | acknowledged | resolved (S-001/S-002 never screen-resolved)
  trigger_text  TEXT,
  label         TEXT,                               -- threshold label, e.g. DESIGN DEFAULT / UNVERIFIED / Device state
  snapshot      TEXT,                               -- JSON object of snapshot values at firing time
  simulator     INTEGER NOT NULL DEFAULT 0
);

CREATE TABLE IF NOT EXISTS work_orders (
  id          TEXT PRIMARY KEY,                     -- export id: wo-<n> within one export
  building_id TEXT NOT NULL REFERENCES buildings(id),
  title       TEXT,
  source      TEXT,
  created     TEXT,
  status      TEXT                                  -- open | complete
);

CREATE TABLE IF NOT EXISTS reports (
  id          TEXT PRIMARY KEY,
  building_id TEXT NOT NULL REFERENCES buildings(id),
  kind        TEXT,                                 -- owner | tech | pack
  created_at  TEXT,
  scan_at     TEXT,                                 -- scannedAt of the scan the report was made from
  codes       TEXT                                  -- JSON array of fault codes in that scan
);

CREATE TABLE IF NOT EXISTS memberships (
  building_id TEXT PRIMARY KEY REFERENCES buildings(id),
  tier        TEXT NOT NULL DEFAULT 'none',         -- none | watch | health | full
  source      TEXT NOT NULL DEFAULT 'manual' CHECK (source IN ('manual','comp')),
  note        TEXT,
  set_at      TEXT                                  -- ISO date the tier was set by hand / comped
);

CREATE INDEX IF NOT EXISTS idx_faults_building_code ON faults(building_id, code);
CREATE INDEX IF NOT EXISTS idx_scans_building_time  ON scans(building_id, scanned_at);
CREATE INDEX IF NOT EXISTS idx_readings_scan        ON readings(scan_id);

// The strip is all-or-nothing: every expected literal is counted first; if any

// Module literals are replaced identically in the .mjs source and in the inlined
// copy inside index.html, so the Part 4 inline-identity tests still pass.

import path from 'node:path';

const target = process.argv[2];
if (!target) { console.error('usage: node strip-public.mjs <target-folder>'); process.exit(2); }

// [file, literal, replacement, expectedCount]  (expectedCount -1 = ">= 10")

const files = {};
for (const f of ['index.html', 'fault-engine.mjs', 'store.mjs', 'gateway.mjs', 'schema.sql']) {
  files[f] = fs.readFileSync(path.join(target, f), 'utf8');
}
const count = (s, lit) => s.split(lit).length - 1;
let bad = 0;
for (const [f, lit, , exp] of RULES) {
  const n = count(files[f], lit);
  const ok = exp === -1 ? n >= 10 : n === exp;
  console.log(`${ok ? 'OK  ' : 'BAD '} ${f}: literal found ${n}x (expected ${exp === -1 ? '>=10' : exp}) — ${lit.slice(0, 60)}`);
  if (!ok) bad++;
}
if (bad) { console.error(`\nSTRIP ABORTED: ${bad} rule(s) mismatched. Nothing written.`); process.exit(1); }
for (const [f, lit, rep] of RULES) files[f] = files[f].split(lit).join(rep);
for (const f of Object.keys(files)) fs.writeFileSync(path.join(target, f), files[f]);
console.log('\nSTRIP APPLIED to', target);
